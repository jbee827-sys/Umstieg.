> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

> **Gewählt (Julian, 16.10.2026):** Ausführung durch Claude selbst (superpowers:executing-plans), nach jedem Paket `/code-review` und `/bug-hunter`, am Ende ein unabhängiger Gesamtreviewer.

**Goal:** Umstieg. zeigt an jeder ÖPNV-Haltestelle in Deutschland den Fußweg vom Ankunfts- zum Abfahrtssteig samt Live-Bewertung des Anschlusses, ohne den bestehenden Bahnhofspfad zu verändern.

**Architecture:** Neben den Bahnhofspfad (EVA, DB-APIs, eigener Graph) tritt ein Haltestellenpfad über die MOTIS-API (heute `api.transitous.org`, Basis-URL ist Konfiguration). Ein Client kapselt MOTIS vollständig und liefert nur eigene Typen; Routen `/api/orte`, `/api/haltestelle/tafel`, `/api/weg`, `/api/live/haltestelle` kommen hinzu. Das Frontend bekommt zwei Ortsfelder (Ankunft, Anschluss); liegen beide im selben DB-Bahnhof, läuft der heutige Ablauf, sonst der Haltestellenpfad mit derselben Weg-Ansicht.

**Tech Stack:** Backend Node ≥ 20, Fastify 5, TypeScript strict, Vitest 3, `lru-cache`, oxlint. Frontend React 19, Vite 8, Vitest + Testing Library, MapLibre.

**Spec:** `docs/superpowers/specs/2026-10-16-umstieg-2-design.md` (Konzept mit F/L/V/A/E-Kennzeichnung) und Befunde `docs/2026-10-16-transitous-erkundung.md` (Paket 0).

```ts
import type { Verkehrsmittel } from '../core/typen.js';

/**
 * MOTIS-Modus → eigenes Verkehrsmittel. Tabelle statt Verzweigung, damit ein
 * neuer Wert eine Zeile ist. Unbekanntes faellt auf `sonstiges` (Spec 5.3) –
 * MOTIS hat bisher drei Werte umbenannt (METRO, AREAL_LIFT, REGIONAL_FAST_RAIL)
 * und wird es wieder tun.
 */
const ZUORDNUNG: Record<string, Verkehrsmittel> = {
  BUS: 'bus',
  COACH: 'bus',
  TRAM: 'tram',
  SUBWAY: 'ubahn',
  METRO: 'ubahn',
  SUBURBAN: 'sbahn',
  RAIL: 'bahn',
  HIGHSPEED_RAIL: 'bahn',
  LONG_DISTANCE: 'bahn',
  NIGHT_RAIL: 'bahn',
  REGIONAL_RAIL: 'bahn',
  REGIONAL_FAST_RAIL: 'bahn',
  FERRY: 'faehre',
};

export function verkehrsmittelAus(mode: string): Verkehrsmittel {
  return ZUORDNUNG[mode] ?? 'sonstiges';
}

/**
 * Linienverkehr mit festen Halten. Mitfahrbaenke (`RIDE_SHARING`), Rufbus-
 * Bereiche (`ODM`, `FLEX`) und Strassenmodi gehoeren nicht dazu (Spec 9:
 * „Rufbus-Bereiche ohne festen Halt nicht“).
 */
const OEV = new Set([
  ...Object.keys(ZUORDNUNG),
  'FUNICULAR',
  'AERIAL_LIFT',
  'AREAL_LIFT',
  'CABLE_CAR',
  'AIRPLANE',
  'OTHER',
]);

export function istOevModus(mode: string): boolean {
  return OEV.has(mode);
}
```

```ts
/**
 * Pausenzustand der MOTIS-API (Spec 5.1, nach dem Muster `overpass-pausen.ts`).
 *
 * Transitous ist ein Gemeinschaftsdienst „best effort“. Wer ihn mit
 * Wiederholungen eindeckt, wird gesperrt – so kam es bei Overpass. Deshalb:
 * Wer 429 sagt, bekommt bis `Retry-After` keine Anfrage; wer gestoert ist
 * (5xx, Zeitueberschreitung, Netz), eine kurze Pause. In der Pause gibt es
 * sofort die Degradation, kein Netzverkehr. Nur im Arbeitsspeicher.
 *
 * Ein Zustand fuer den ganzen Anbieter, nicht je Endpunkt: Es ist ein Host,
 * und ein 429 von der Suche gilt fuer die Tafel genauso.
 */
export const PAUSE_429_STANDARD_MS = 60_000;
export const PAUSE_STOERUNG_MS = 30_000;
/** Ein `Retry-After` von einem Tag wuerde den Pfad unbemerkt abschalten. */
export const PAUSE_HOECHSTENS_MS = 15 * 60_000;

/** Pause, Zeitueberschreitung, 5xx, Netz: wiederkommen hilft. */
export class HaltestellenNichtVerfuegbarError extends Error {
  constructor(grund: string) {
    super(`Haltestellendaten nicht verfuegbar: ${grund}`);
    this.name = 'HaltestellenNichtVerfuegbarError';
  }
}

/** 4xx ausser 429: Die Anfrage selbst passt nicht (unbekannte Fahrt, kaputte ID). */
export class HaltestellenAnfrageFehler extends Error {
  constructor(readonly status: number, detail: string) {
    super(`MOTIS HTTP ${status}: ${detail.slice(0, 160)}`);
    this.name = 'HaltestellenAnfrageFehler';
  }
}

export function retryAfterMs(kopf: string | null | undefined, jetzt: Date): number | null {
  if (!kopf) return null;
  const roh = kopf.trim();
  if (/^\d+$/.test(roh)) return Number(roh) * 1000;
  const datum = Date.parse(roh);
  if (Number.isNaN(datum)) return null;
  return Math.max(0, datum - jetzt.getTime());
}

export interface MotisPausen {
  pausiertBis(): Date | null;
  merke(e: { status?: number; retryAfter?: string | null; fehler?: unknown }): number;
}

export function erzeugeMotisPausen(o: { jetzt?: () => Date } = {}): MotisPausen {
  const jetzt = o.jetzt ?? (() => new Date());
  let bis = 0;
  return {
    pausiertBis() {
      if (bis <= jetzt().getTime()) return null;
      return new Date(bis);
    },
    merke({ status, retryAfter, fehler }) {
      let ms = 0;
      if (status === 429) {
        ms = Math.min(retryAfterMs(retryAfter, jetzt()) ?? PAUSE_429_STANDARD_MS, PAUSE_HOECHSTENS_MS);
      } else if ((status !== undefined && status >= 500) || (status === undefined && fehler !== undefined)) {
        ms = PAUSE_STOERUNG_MS;
      }
      if (ms > 0) bis = Math.max(bis, jetzt().getTime() + ms);
      return ms;
    },
  };
}
```

```ts
/**
 * Anfragen an die MOTIS-API je Endpunkt und Stunde (Spec 5.4, K7).
 *
 * Der Nachweis fuer die Absprache mit Transitous: nicht geschaetzt, sondern
 * gezaehlt. Gezaehlt wird, was hinausgeht – keine Treffer aus dem
 * Zwischenspeicher, keine Anfragen, die die Pause abgefangen hat. Protokolliert
 * werden Summen, nie wer gefragt hat.
 */
export interface ZaehlerProtokoll {
  info(daten: Record<string, unknown>, nachricht: string): void;
}

export interface Anfragezaehler {
  zaehle(endpunkt: string): void;
  schreibe(): void;
  stand(): Record<string, number>;
  beende(): void;
}

export function erzeugeAnfragezaehler(
  o: { log?: ZaehlerProtokoll; jetzt?: () => Date; intervallMs?: number } = {},
): Anfragezaehler {
  const jetzt = o.jetzt ?? (() => new Date());
  let je: Record<string, number> = {};
  const zaehler: Anfragezaehler = {
    zaehle(endpunkt) {
      je[endpunkt] = (je[endpunkt] ?? 0) + 1;
    },
    schreibe() {
      const summe = Object.values(je).reduce((s, n) => s + n, 0);
      if (summe === 0) return;
      o.log?.info(
        { ereignis: 'motis-anfragen', stunde: jetzt().toISOString().slice(0, 13), je, summe },
        'MOTIS-Anfragen der Stunde',
      );
      je = {};
    },
    stand: () => ({ ...je }),
    beende() {
      if (timer) clearInterval(timer);
    },
  };
  const intervall = o.intervallMs ?? 60 * 60_000;
  const timer = intervall > 0 ? setInterval(() => zaehler.schreibe(), intervall) : undefined;
  timer?.unref?.();
  return zaehler;
}
```

```ts
import type { Anfragezaehler } from '../core/anfragezaehler.js';
import type { Steigpunkt } from '../core/typen.js';
import { APP_VERSION } from '../version.js';
import {
  HaltestellenAnfrageFehler,
  HaltestellenNichtVerfuegbarError,
  type MotisPausen,
} from './motis-pausen.js';
import type { MotisItinerary, MotisMatch, MotisPlanAntwort, MotisStoptimesAntwort } from './motis-typen.js';

/**
 * Endpunkte der MOTIS-API. MOTIS versioniert je Endpunkt (heute v1 und v6);
 * ein Versionswechsel betrifft genau diese Stelle (Spec 5.1).
 */
export const MOTIS_PFADE = {
  geocode: '/api/v1/geocode',
  reverseGeocode: '/api/v1/reverse-geocode',
  stoptimes: '/api/v6/stoptimes',
  trip: '/api/v6/trip',
  plan: '/api/v6/plan',
} as const;

/** Gemessen am 16.10.2026: Maximum 299 ms. Die Grenzen sind Schutz, kein Ziel. */
export const MOTIS_ZEITGRENZEN_MS = { suche: 4000, tafel: 4000, fussweg: 8000, fahrt: 4000 } as const;

/**
 * Zwei Stunden ab dem Anfragezeitpunkt (der Client fragt ab jetzt − 5 min).
 * Das breite Layout zeigt bis zum Ende der naechsten vollen Stunde – das sind
 * hoechstens zwei Stunden.
 */
const TAFEL_FENSTER_S = 2 * 60 * 60;
/**
 * Umkreis fuer den Rueckfall ueber die Koordinate (A5): Ist die ID nach einem
 * Datenwechsel unbekannt, findet MOTIS die Haltestelle ueber `center`. 30 m
 * reichen fuer den Steig selbst; aequivalente Halte nimmt MOTIS ohnehin dazu.
 */
const RUECKFALL_RADIUS_M = 30;

export function motisUserAgent(kontakt: string | undefined): string {
  const zusatz = kontakt?.trim() ? `; ${kontakt.trim()}` : '';
  return `Umstieg/${APP_VERSION} (+https://www.umstieg.online${zusatz})`;
}

/** Drei Nachkommastellen, rund 100 m (Spec 6, Datenschutz). */
export function rundeKoordinate(wert: number): number {
  return Math.round(wert * 1000) / 1000;
}

function punkt(p: Steigpunkt): string {
  return p.level === undefined ? `${p.lat},${p.lon}` : `${p.lat},${p.lon},${p.level}`;
}

export interface MotisKern {
  geocode(text: string): Promise<MotisMatch[]>;
  reverseGeocode(lat: number, lon: number): Promise<MotisMatch[]>;
  stoptimes(a: { stopId: string; lat: number; lon: number; typ: 'arr' | 'dep'; ab: Date }): Promise<MotisStoptimesAntwort>;
  trip(tripId: string): Promise<MotisItinerary>;
  plan(a: { von: Steigpunkt; nach: Steigpunkt; profil: 'FOOT' | 'WHEELCHAIR' }): Promise<MotisPlanAntwort>;
}

export function erzeugeMotisKern(o: {
  basisUrl: string;
  kontakt?: string;
  fetchFn?: typeof fetch;
  pausen?: MotisPausen;
  zaehler?: Anfragezaehler;
}): MotisKern {
  const basis = o.basisUrl.replace(/\/+$/, '');
  const fetchFn = o.fetchFn ?? fetch;
  const userAgent = motisUserAgent(o.kontakt);

  async function hole<T>(
    name: keyof typeof MOTIS_PFADE,
    parameter: Record<string, string>,
    zeitgrenzeMs: number,
  ): Promise<T> {
    const pause = o.pausen?.pausiertBis();
    if (pause) throw new HaltestellenNichtVerfuegbarError(`pausiert bis ${pause.toISOString()}`);

    const url = `${basis}${MOTIS_PFADE[name]}?${new URLSearchParams(parameter).toString()}`;
    o.zaehler?.zaehle(name);
    let antwort: Response;
    try {
      antwort = await fetchFn(url, {
        headers: { 'User-Agent': userAgent, Accept: 'application/json' },
        signal: AbortSignal.timeout(zeitgrenzeMs),
      });
    } catch (fehler) {
      o.pausen?.merke({ fehler });
      throw new HaltestellenNichtVerfuegbarError(String(fehler).slice(0, 160));
    }
    if (!antwort.ok) {
      const text = await antwort.text().catch(() => '');
      o.pausen?.merke({ status: antwort.status, retryAfter: antwort.headers.get('retry-after') });
      if (antwort.status === 429 || antwort.status >= 500) {
        throw new HaltestellenNichtVerfuegbarError(`HTTP ${antwort.status}`);
      }
      throw new HaltestellenAnfrageFehler(antwort.status, text);
    }
    try {
      return (await antwort.json()) as T;
    } catch (fehler) {
      o.pausen?.merke({ fehler });
      throw new HaltestellenNichtVerfuegbarError('Antwort nicht lesbar');
    }
  }

  return {
    geocode: (text) =>
      hole<MotisMatch[]>('geocode', { text, type: 'STOP', language: 'de' }, MOTIS_ZEITGRENZEN_MS.suche),
    reverseGeocode: (lat, lon) =>
      hole<MotisMatch[]>(
```

```ts
const UMLAUTE: Record<string, string> = { ä: 'ae', ö: 'oe', ü: 'ue', ß: 'ss' };
/** Vorsilben und Bahnhofswoerter - dieselben Klassen wie `bahnhofsname.ts`. */
const STOERWORTE = new Set(['s', 'u', 's+u', 'u+s', 'bus', 'tram', 'str', 'bahnhof', 'bhf', 'bf']);

export function namensSchluessel(name: string): string {
  const klein = name
    .toLowerCase()
    .replace(/[äöüß]/g, (z) => UMLAUTE[z])
    .replace(/\([^)]*\)/g, ' ')
    .replace(/hauptbahnhof/g, 'hbf');
  return klein
    .split(/[^a-z0-9+]+/)
    .filter((w) => w && !STOERWORTE.has(w))
    .join(' ');
}
```

```ts
import type { Ort, Verkehrsmittel } from '../core/typen.js';
import { namensSchluessel } from '../core/orte.js';
import type { MotisMatch } from './motis-typen.js';

const REIHENFOLGE: Verkehrsmittel[] = ['bahn', 'sbahn', 'ubahn', 'tram', 'bus', 'faehre', 'sonstiges'];
/** Unter diesem Abstand ist derselbe Name derselbe Halt aus einer zweiten Quelle. */
const DOPPELT_M = 150;

export function abstandM(a: { lat: number; lon: number }, b: { lat: number; lon: number }): number {
  const r = 6_371_000;
  const rad = Math.PI / 180;
  const dLat = (b.lat - a.lat) * rad;
  const dLon = (b.lon - a.lon) * rad;
  const h = Math.sin(dLat / 2) ** 2 + Math.cos(a.lat * rad) * Math.cos(b.lat * rad) * Math.sin(dLon / 2) ** 2;
  return 2 * r * Math.asin(Math.sqrt(h));
}

export function verkehrsmittelListe(modes: readonly string[] | undefined): Verkehrsmittel[] {
  const da = new Set((modes ?? []).filter(istOevModus).map(verkehrsmittelAus));
  return REIHENFOLGE.filter((v) => da.has(v));
}

/**
 * Suchtreffer → Orte. Gefiltert wird, was der Erkundungsbericht vom 16.10.2026
 * gefunden hat: fremde Laender (Wien, New Hamburg), Mitfahrbaenke und
 * Parkplaetze ohne Linienverkehr, derselbe Halt aus zwei Quellen.
 */
export function orteAusTreffern(treffer: MotisMatch[]): Ort[] {
  const orte: Ort[] = [];
  for (const t of Array.isArray(treffer) ? treffer : []) {
    if (t.type !== 'STOP') continue;
    if (t.country !== undefined && t.country !== 'DE') continue;
    const verkehrsmittel = verkehrsmittelListe(t.modes);
    if (verkehrsmittel.length === 0) continue;
    const schluessel = namensSchluessel(t.name);
    const doppelt = orte.some(
      (o) => namensSchluessel(o.name) === schluessel && abstandM(o, t) <= DOPPELT_M,
    );
    if (doppelt) continue;
    orte.push({ name: t.name, lat: t.lat, lon: t.lon, verkehrsmittel, haltestelleId: t.id });
  }
  return orte;
}
```

```ts
import type { Fahrtverlauf, HaltestellenTafelEintrag } from '../core/typen.js';
import type { MotisItinerary, MotisPlace, MotisStopTime, MotisStoptimesAntwort } from './motis-typen.js';

/**
 * Steig aus einer DHID-Steigkennung (VDV 432): `de:<Kreis>:<Nr>:<Bereich>:<Steig>`.
 * Gelesen wird NUR zur Anzeige, nie als Schluessel (Lockerung von Spec 5.1,
 * begruendet im Erkundungsbericht A2). Alles andere ergibt `undefined`.
 */
const DHID_STEIG = /(?:^|_)de:\d{5}:[^:_]+:[^:_]*:([^:_]+)$/;
export function steigAusDhid(stopId: string | undefined): string | undefined {
  return stopId ? (DHID_STEIG.exec(stopId)?.[1] ?? undefined) : undefined;
}

function eintragAus(s: MotisStopTime, typ: 'arr' | 'dep'): HaltestellenTafelEintrag | null {
  const p: MotisPlace = s.place;
  const zeitSoll = typ === 'dep' ? p.scheduledDeparture : p.scheduledArrival;
  if (!zeitSoll) return null;
  const zeitIst = s.realTime ? (typ === 'dep' ? p.departure : p.arrival) : undefined;
  const dhid = steigAusDhid(p.stopId);
  const steigSoll = p.scheduledTrack ?? p.track ?? dhid;
  const steigIst = s.realTime && p.track && p.track !== steigSoll ? p.track : undefined;
  return {
    fahrtId: s.tripId,
    linie: s.displayName || s.routeShortName || s.tripShortName || '',
    verkehrsmittel: verkehrsmittelAus(s.mode),
    ziel: (typ === 'dep' ? s.headsign || s.tripTo?.name : s.tripFrom?.name) ?? '',
    zeitSoll,
    ...(zeitIst ? { zeitIst } : {}),
    ...(steigSoll ? { steigSoll } : {}),
    ...(steigIst ? { steigIst } : {}),
    steigLat: p.lat,
    steigLon: p.lon,
    ...(typeof p.level === 'number' ? { steigLevel: p.level } : {}),
    steigGenau: Boolean(p.track || p.scheduledTrack || p.parentId || dhid),
    echtzeit: s.realTime === true,
    ausfall: s.cancelled === true || s.tripCancelled === true,
  };
}

/** Vorrang beim Entdoppeln: genauer Steig vor ungenauem, Echtzeit vor Fahrplan. */
function rang(e: HaltestellenTafelEintrag): number {
  return (e.steigGenau ? 2 : 0) + (e.echtzeit ? 1 : 0);
}

export function tafelAusStoptimes(antwort: MotisStoptimesAntwort, typ: 'arr' | 'dep'): HaltestellenTafelEintrag[] {
  const je = new Map<string, HaltestellenTafelEintrag>();
  for (const s of Array.isArray(antwort?.stopTimes) ? antwort.stopTimes : []) {
    if (!istOevModus(s.mode)) continue;
    // Ein Ausfall traegt ebenfalls NOT_ALLOWED - er muss sichtbar bleiben.
    const ausfall = s.cancelled === true || s.tripCancelled === true;
    if (s.pickupDropoffType === 'NOT_ALLOWED' && !ausfall) continue;
    const e = eintragAus(s, typ);
    if (!e) continue;
    const schluessel = `${e.linie}|${e.ziel}|${e.zeitSoll}`;
    const vorher = je.get(schluessel);
    if (!vorher) {
      je.set(schluessel, e);
      continue;
    }
    // Dieselbe Fahrt aus zwei Quellen (DELFI und VBB, Wiesenburg 16.10.2026).
    // Meldet EINE einen Ausfall, wird gewarnt: lieber einmal zu vorsichtig
    // als ein verpasster Hinweis.
    const besser = rang(e) > rang(vorher) ? e : vorher;
    je.set(schluessel, { ...besser, ausfall: vorher.ausfall || e.ausfall });
  }
  return [...je.values()].sort((a, b) => (a.zeitIst ?? a.zeitSoll).localeCompare(b.zeitIst ?? b.zeitSoll));
}

export function fahrtverlaufAusTrip(it: MotisItinerary): Fahrtverlauf {
  const leg = it.legs.find((l) => istOevModus(l.mode)) ?? it.legs[0];
  const halte = [leg.from, ...(leg.intermediateStops ?? []), leg.to].map((p) => ({
    name: p.name,
    ...(p.scheduledDeparture ?? p.scheduledArrival ? { zeitSoll: p.scheduledDeparture ?? p.scheduledArrival } : {}),
    ...(p.departure ?? p.arrival ? { zeitIst: p.departure ?? p.arrival } : {}),
    ...(p.track ?? steigAusDhid(p.stopId) ? { steig: p.track ?? steigAusDhid(p.stopId) } : {}),
  }));
  return { linie: leg.displayName || leg.routeShortName || '', ziel: leg.headsign || leg.to.name, halte };
}
```

```ts
import type { GraphKante, GraphKnoten, KantenTyp, RoutenPfad } from '../core/typen.js';
import { abstandM } from './motis-abbildung.js';
import type { MotisPlanAntwort, MotisSchritt } from './motis-typen.js';

/** Google-Polylinie mit wählbarer Genauigkeit (MOTIS v6: 6 Stellen). */
export function dekodierePolylinie(punkte: string, genauigkeit: number): { lat: number; lon: number }[] {
  const faktor = 10 ** genauigkeit;
  const ergebnis: { lat: number; lon: number }[] = [];
  let i = 0;
  let lat = 0;
  let lon = 0;
  const naechste = (): number => {
    let wert = 0;
    let schiebe = 0;
    let byte: number;
    do {
      byte = punkte.charCodeAt(i++) - 63;
      wert |= (byte & 0x1f) << schiebe;
      schiebe += 5;
    } while (byte >= 0x20 && i < punkte.length + 1);
    return wert & 1 ? ~(wert >> 1) : wert >> 1;
  };
  while (i < punkte.length) {
    lat += naechste();
    lon += naechste();
    ergebnis.push({ lat: Math.round((lat / faktor) * 1e6) / 1e6, lon: Math.round((lon / faktor) * 1e6) / 1e6 });
  }
  return ergebnis;
}

const IST_WECHSEL = new Set(['STAIRS', 'ELEVATOR']);

/** Ebene, auf der der Weg nach Schritt `i` weitergeht - aus dem naechsten ebenen Schritt. */
function ebeneDanach(schritte: MotisSchritt[], i: number, vorher: number): number {
  for (let j = i + 1; j < schritte.length; j++) {
    if (!IST_WECHSEL.has(schritte[j].relativeDirection)) return schritte[j].fromLevel ?? 0;
  }
  const s = schritte[i];
  // Letzter Schritt ist der Wechsel selbst: der Wert, der NICHT der Ausgang ist.
  return s.fromLevel === vorher ? (s.toLevel ?? vorher) : (s.fromLevel ?? vorher);
}

/**
 * Fussweg aus `/plan` → derselbe `RoutenPfad`, den der Bahnhofspfad aus seinem
 * Graphen baut. Damit rechnen Gehzeit (`core/gehzeit.ts`) und Schrittsaetze
 * (`core/textgen.ts`) auf beiden Pfaden gleich (Spec 5.5).
 *
 * Zwei Fallen aus der Erkundung (16.10.2026): MOTIS liefert keine
 * Abbiegerichtung - die rechnet textgen aus der Geometrie -, und die Ebenen
 * einer Treppe stehen teils gegen die Gehrichtung. Deshalb kommt die Richtung
 * aus den Schritten davor und danach.
 */
export function pfadAusPlan(antwort: MotisPlanAntwort): RoutenPfad | null {
  const leg = antwort?.direct?.[0]?.legs?.find((l) => l.mode === 'WALK');
  const schritte = leg?.steps ?? [];
  if (!leg || schritte.length === 0) return null;

  const knoten: GraphKnoten[] = [];
  const kanten: GraphKante[] = [];
  const neuerKnoten = (lat: number, lon: number, level: number): GraphKnoten => {
    const k = { id: `w${knoten.length}`, lat, lon, level };
    knoten.push(k);
    return k;
  };
  const verbinde = (nach: GraphKnoten, typ: KantenTyp, laengeM: number): void => {
    const von = knoten[knoten.length - 2];
    kanten.push({
      von: von.id,
      nach: nach.id,
      laengeM,
      deltaLevel: nach.level - von.level,
      typ,
      barrierefrei: typ !== 'treppe',
    });
  };
  const start = leg.from;
  neuerKnoten(start.lat, start.lon, start.level ?? schritte[0].fromLevel ?? 0);

  schritte.forEach((s, i) => {
    const letzter = knoten[knoten.length - 1];
    const punkte = dekodierePolylinie(s.polyline?.points ?? '', s.polyline?.precision ?? 6);
    if (IST_WECHSEL.has(s.relativeDirection)) {
      const nachher = ebeneDanach(schritte, i, letzter.level);
      const ende = punkte.at(-1) ?? letzter;
      if (nachher === letzter.level) {
        if (abstandM(letzter, ende) >= 0.5) {
          verbinde(neuerKnoten(ende.lat, ende.lon, letzter.level), 'eben', abstandM(letzter, ende));
        }
        return;
      }
      const typ: KantenTyp = s.relativeDirection === 'ELEVATOR' ? 'aufzug' : 'treppe';
      verbinde(neuerKnoten(ende.lat, ende.lon, nachher), typ, typ === 'aufzug' ? 0 : s.distance);
      return;
    }
    const level = s.fromLevel === s.toLevel ? (s.fromLevel ?? letzter.level) : letzter.level;
    for (const p of punkte) {
      const vorher = knoten[knoten.length - 1];
      if (abstandM(vorher, p) < 0.5 && vorher.level === level) continue;
      verbinde(neuerKnoten(p.lat, p.lon, level), vorher.level === level ? 'eben' : 'treppe', abstandM(vorher, p));
    }
  });

  if (kanten.length === 0) return null;
  return { knoten, kanten, laengeM: kanten.reduce((s, k) => s + k.laengeM, 0) };
}
```

```ts
import type { Fahrtverlauf, HaltestellenTafelEintrag, Ort, RoutenPfad, Steigpunkt } from '../core/typen.js';

import { pfadAusPlan } from './motis-weg.js';

export const HALTESTELLEN_SUCHE_TTL_MS = 24 * 60 * 60_000;
/** Ueber dem Bahnhofswert (20 s), weil der Live-Abstand hier 60 s ist, nicht 30 s. */
export const HALTESTELLEN_TAFEL_TTL_MS = 30_000;
export const FUSSWEG_TTL_MS = 24 * 60 * 60_000;
export const FAHRT_TTL_MS = 60_000;
/** Passt zu FENSTER_VOR_MIN im Frontend: wer gerade weg ist, ist meist noch da. */
const TAFEL_VORLAUF_MS = 5 * 60_000;

export interface HaltestelleAnfrage {
  id: string;
  lat: number;
  lon: number;
}

export interface HaltestellenClient {
  sucheOrte(text: string): Promise<Ort[]>;
  orteNahebei(lat: number, lon: number): Promise<Ort[]>;
  tafel(h: HaltestelleAnfrage, typ: 'arr' | 'dep'): Promise<HaltestellenTafelEintrag[]>;
  tafelFrisch(h: HaltestelleAnfrage, typ: 'arr' | 'dep'): Promise<HaltestellenTafelEintrag[]>;
  fussweg(von: Steigpunkt, nach: Steigpunkt, profil: 'FOOT' | 'WHEELCHAIR'): Promise<RoutenPfad | null>;
  fahrt(fahrtId: string): Promise<Fahrtverlauf>;
}

const punktSchluessel = (p: Steigpunkt): string => `${p.lat.toFixed(5)},${p.lon.toFixed(5)},${p.level ?? ''}`;

export function erzeugeHaltestellenClient(kern: MotisKern, o: { jetzt?: () => Date } = {}): HaltestellenClient {
  const jetzt = o.jetzt ?? (() => new Date());
  const tafelFrisch = async (h: HaltestelleAnfrage, typ: 'arr' | 'dep') =>
    tafelAusStoptimes(
      await kern.stoptimes({ stopId: h.id, lat: h.lat, lon: h.lon, typ, ab: new Date(jetzt().getTime() - TAFEL_VORLAUF_MS) }),
      typ,
    );
  return {
    sucheOrte: mitCache(async (text: string) => orteAusTreffern(await kern.geocode(text)), (t) => t.trim().toLowerCase(), HALTESTELLEN_SUCHE_TTL_MS, 500),
    // BEWUSST OHNE Zwischenspeicher - der Schluessel waere die Koordinate des
    // Nutzers (gleiche Begruendung wie `sucheBahnhoefeNahebei` in app.ts).
    orteNahebei: async (lat, lon) => orteAusTreffern(await kern.reverseGeocode(lat, lon)),
    tafel: mitCache(tafelFrisch, (h, typ) => `${h.id}|${typ}`, HALTESTELLEN_TAFEL_TTL_MS, 300),
    tafelFrisch,
    fussweg: mitCache(
      async (von: Steigpunkt, nach: Steigpunkt, profil: 'FOOT' | 'WHEELCHAIR') => pfadAusPlan(await kern.plan({ von, nach, profil })),
      (von, nach, profil) => `${punktSchluessel(von)}|${punktSchluessel(nach)}|${profil}`,
      FUSSWEG_TTL_MS,
      500,
    ),
    fahrt: mitCache(async (id: string) => fahrtverlaufAusTrip(await kern.trip(id)), (id) => id, FAHRT_TTL_MS, 200),
  };
}
```

```ts
import { abstandM } from '../clients/motis-abbildung.js';
import type { BahnhofTreffer, Ort } from './typen.js';

export const GLEICHER_ORT_M = 400;

export function bahnhofAlsOrt(b: BahnhofTreffer): Ort {
  return { name: b.name, eva: b.eva, lat: b.lat, lon: b.lon, verkehrsmittel: ['bahn'] };
}

function namenPassen(a: string, b: string): boolean {
  const wa = new Set(namensSchluessel(a).split(' ').filter(Boolean));
  const wb = new Set(namensSchluessel(b).split(' ').filter(Boolean));
  if (wa.size === 0 || wb.size === 0) return false;
  const [klein, gross] = wa.size <= wb.size ? [wa, wb] : [wb, wa];
  return [...klein].every((w) => gross.has(w));
}

/**
 * Bahnhoefe (RIS::Stations) und Haltestellen (MOTIS) zu einer Trefferliste.
 *
 * Derselbe Ort, wenn die Haltestelle Bahnverkehr hat, hoechstens 400 m entfernt
 * liegt und der Name passt. Die Bahnverkehr-Pflicht verhindert, dass die
 * Bushaltestelle vor dem Bahnhof die EVA erbt und damit in den Bahnhofspfad
 * faellt, der sie nicht kennt.
 */
export function fuehreOrteZusammen(bahnhoefe: BahnhofTreffer[], haltestellen: Ort[], max = 8): Ort[] {
  const vergeben = new Set<Ort>();
  const ergebnis: Ort[] = [];
  for (const b of bahnhoefe) {
    const passend = haltestellen.find(
      (h) =>
        !vergeben.has(h) &&
        (h.verkehrsmittel.includes('bahn') || h.verkehrsmittel.includes('sbahn')) &&
        abstandM(h, b) <= GLEICHER_ORT_M &&
        namenPassen(h.name, b.name),
    );
    if (passend) {
      vergeben.add(passend);
      ergebnis.push({ ...bahnhofAlsOrt(b), verkehrsmittel: passend.verkehrsmittel, haltestelleId: passend.haltestelleId });
    } else {
      ergebnis.push(bahnhofAlsOrt(b));
    }
  }
  for (const h of haltestellen) if (!vergeben.has(h)) ergebnis.push(h);
  return ergebnis.slice(0, max);
}
```

```ts
export async function sucheBahnhoefeMitVarianten(stations: StationsClient, q: string): Promise<BahnhofTreffer[]> {
```

```ts
import type { HaltestellenClient } from '../clients/haltestellen.js';
import { FehlendeZugangsdatenError } from '../clients/stations.js';
import { fuehreOrteZusammen } from '../core/orte.js';
import type { BahnhofTreffer, Ort } from '../core/typen.js';
import { leseKoordinaten, sucheBahnhoefeMitVarianten, type StationsClient } from './api.js';

export type QuellenStand = 'ok' | 'zugang-fehlt' | 'ausfall' | 'nicht-eingerichtet';
export interface OrteAntwort {
  orte: Ort[];
  quellen: { bahnhoefe: QuellenStand; haltestellen: QuellenStand };
}
export interface HaltestellenRoutenOptionen {
  stations: StationsClient;
  haltestellen: HaltestellenClient | null;
}

const ERKENNUNG_RADIUS_M = 1000;

export function registriereHaltestellenRouten(app: FastifyInstance, o: HaltestellenRoutenOptionen): void {
  app.get<{ Querystring: { q?: string; lat?: string; lon?: string } }>('/api/orte', async (req, reply) => {
    const q = req.query.q?.trim();
    const hatKoordinaten = req.query.lat !== undefined || req.query.lon !== undefined;
    if (!q && !hatKoordinaten) return reply.code(400).send({ fehler: 'parameter-fehlt', detail: 'q oder lat+lon ist erforderlich' });
    if (q && hatKoordinaten) return reply.code(400).send({ fehler: 'parameter-ungueltig', detail: 'q und lat/lon schliessen sich aus' });
    const punkt = hatKoordinaten ? leseKoordinaten(req.query.lat, req.query.lon) : null;
    if (hatKoordinaten && !punkt) return reply.code(400).send({ fehler: 'parameter-ungueltig', detail: 'lat und lon muessen gueltig sein' });

    const [db, motis] = await Promise.allSettled([
      punkt ? o.stations.sucheBahnhoefeNahebei(punkt.lat, punkt.lon, ERKENNUNG_RADIUS_M) : sucheBahnhoefeMitVarianten(o.stations, q!),
      o.haltestellen ? (punkt ? o.haltestellen.orteNahebei(punkt.lat, punkt.lon) : o.haltestellen.sucheOrte(q!)) : Promise.resolve(null),
    ]);
    const bahnhoefe: BahnhofTreffer[] = db.status === 'fulfilled' ? db.value : [];
    const haltestellen: Ort[] = motis.status === 'fulfilled' ? (motis.value ?? []) : [];
    const quellen: OrteAntwort['quellen'] = {
      bahnhoefe: db.status === 'fulfilled' ? 'ok' : db.reason instanceof FehlendeZugangsdatenError ? 'zugang-fehlt' : 'ausfall',
      haltestellen: !o.haltestellen ? 'nicht-eingerichtet' : motis.status === 'fulfilled' ? 'ok' : 'ausfall',
    };
    if (motis.status === 'rejected') {
      req.log.warn({ degradation: 'haltestellen-unerreichbar', route: '/api/orte', grund: String(motis.reason) }, 'Haltestellensuche ohne MOTIS - nur Bahnhoefe');
    }
    if (db.status === 'rejected' && !(db.reason instanceof FehlendeZugangsdatenError)) {
      req.log.warn({ degradation: 'bahnhofssuche-unerreichbar', route: '/api/orte', grund: String(db.reason) }, 'Ortssuche ohne RIS::Stations');
    }
    const keineQuelle = quellen.bahnhoefe !== 'ok' && quellen.haltestellen !== 'ok';
    if (keineQuelle && !(quellen.bahnhoefe === 'zugang-fehlt' && quellen.haltestellen === 'nicht-eingerichtet')) {
      return reply.code(503).send({ fehler: 'orte-nicht-verfuegbar', detail: 'Die Ortssuche ist gerade nicht erreichbar.', quellen });
    }
    return { orte: fuehreOrteZusammen(bahnhoefe, haltestellen), quellen } satisfies OrteAntwort;
  });
}
```

```ts
export const UMWEGFAKTOR = 1.3;

export function wegAusPfad(pfad: RoutenPfad, profil: Gehprofil, zielZusatz?: string): Weg {
```

```ts
    stufe: 1,
    schritte: generiereSchritte(pfad, { profil, ...(zielZusatz ? { zielZusatz } : {}) }),

export function wegAusLuftlinie(von: Steigpunkt, nach: Steigpunkt, profil: Gehprofil): Weg {
```
